# Swimming club, FAULTY VARIANT (an instructor may quit after requesting work): one season, two teams.
# Transition names carry an agent prefix (a_, i_, s_) because names must be unique
# across all agents; the label (as in the figure) is what synchronises.
# move_1 moves a swimmer from team 1 to team 2, move_2 from team 2 to team 1.
 
# ---------------------------------------------------------------- Admin
agent: Admin
  places: a_plan, a_season, season_end
  transitions: a_form_teams, a_close_season
               a_join_team_1, a_join_team_2, a_mail_1, a_mail_2
               a_move_1, a_move_2, a_end_1, a_end_2
  labels: a_form_teams=form_teams, a_close_season=close_season
          a_join_team_1=join_team_1, a_join_team_2=join_team_2
          a_mail_1=mail_1, a_mail_2=mail_2
          a_move_1=move_1, a_move_2=move_2
          a_end_1=end_1, a_end_2=end_2
  arcs: (a_plan, a_form_teams) (a_form_teams, a_season)
        (a_season, a_close_season) (a_close_season, season_end)
  read: (a_season, a_join_team_1) (a_season, a_join_team_2)
        (a_season, a_mail_1) (a_season, a_mail_2)
        (a_season, a_move_1) (a_season, a_move_2)
